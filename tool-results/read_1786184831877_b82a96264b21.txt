    84→const fmtDateTime = (iso: string) =>
    85→  new Date(iso).toLocaleDateString('en-KE', {

    88→  })

    90→const fmtEventDate = (iso: string, endDate: string | null) => {
    91→  const d = new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short' })

    93→    const e = new Date(endDate).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })

    96→  return d
