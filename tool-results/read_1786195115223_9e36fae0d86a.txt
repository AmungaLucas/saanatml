    93→const fmtDateTime = (iso: string) =>

    99→const fmtEventDate = (iso: string, endDate: string | null) => {
   100→  const d = new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short' })
   101→  if (endDate) {
   102→    const e = new Date(endDate).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
   103→    return `${d} – ${e}`
   104→  }
   105→  return d
   106→}

   108→const timeAgo = (iso: string) => {
   109→  const seconds = Math.floor((Date.now() - new Date(iso).getTime()) / 1000)
   110→  if (seconds < 60) return 'just now'
   111→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   112→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   113→  return `${Math.floor(seconds / 86400)}d ago`
