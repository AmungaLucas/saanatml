    86→const slugify = (text: string) => text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, '')

    88→const toDatetimeLocal = (iso: string) => {
    89→  if (!iso) return ''
    90→  const d = new Date(iso)
    91→  const pad = (n: number) => String(n).padStart(2, '0')
    92→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    93→}
    94→
    95→const fmtDate = (iso: string) => new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    96→const fmtDateTime = (iso: string) => new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
    97→const timeAgo = (iso: string) => {
    98→  const seconds = Math.floor((Date.now() - new Date(iso).getTime()) / 1000)
    99→  if (seconds < 60) return 'just now'
   100→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   101→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   102→  return `${Math.floor(seconds / 86400)}d ago`
   103→}
