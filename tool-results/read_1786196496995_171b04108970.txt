    87→// ── Helpers ─────────────────────────────────────────────────
    88→
    89→const slugify = (text: string) => text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, '')
    90→
    91→const fmtDate = (iso: string) =>
    92→  new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    93→
    94→const fmtDateTime = (iso: string) =>
    95→  new Date(iso).toLocaleDateString('en-KE', {
    96→    day: 'numeric', month: 'short', year: 'numeric',
    97→    hour: '2-digit', minute: '2-digit',
    98→  })
    99→
   100→const fmtEventDate = (iso: string, endDate: string | null) => {
   101→  const d = new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short' })
   102→  if (endDate) {
   103→    const e = new Date(endDate).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
   104→    return `${d} – ${e}`
   105→  }
   106→  return d
   107→}
   108→
   109→const timeAgo = (iso: string) => {
   110→  const seconds = Math.floor((Date.now() - new Date(iso).getTime()) / 1000)
   111→  if (seconds < 60) return 'just now'
   112→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   113→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   114→  return `${Math.floor(seconds / 86400)}d ago`
   115→}
