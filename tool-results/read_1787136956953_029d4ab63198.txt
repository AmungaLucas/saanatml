    78→// ── Helpers ─────────────────────────────────────────────────
    79→
    80→const slugify = (text: string) => text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, '')
    81→
    82→const fmtDate = (iso: string) =>
    83→  new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    84→
    85→const fmtDateTime = (iso: string) =>
    86→  new Date(iso).toLocaleDateString('en-KE', {
    87→    day: 'numeric', month: 'short', year: 'numeric',
    88→    hour: '2-digit', minute: '2-digit',
    89→  })
    90→
    91→const fmtEventDate = (iso: string, endDate: string | null) => {
    92→  const d = new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short' })
    93→  if (endDate) {
    94→    const e = new Date(endDate).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    95→    return `${d} – ${e}`
    96→  }
    97→  return d
    98→}
    99→
   100→const timeAgo = (iso: string) => {
   101→  const seconds = Math.floor((Date.now() - new Date(iso).getTime()) / 1000)
   102→  if (seconds < 60) return 'just now'
   103→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   104→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   105→  return `${Math.floor(seconds / 86400)}d ago`
   106→}
