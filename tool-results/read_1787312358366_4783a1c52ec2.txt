    87→const slugify = (text: string) => text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, '')
    88→
    89→const toDatetimeLocal = (iso: string) => {
    90→  if (!iso) return ''
    91→  const d = new Date(iso)
    92→  const pad = (n: number) => String(n).padStart(2, '0')
    93→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    94→}
    95→
    96→const fmtDate = (iso: string) => new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    97→const fmtDateTime = (iso: string) => new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
    98→const timeAgo = (iso: string) => {
    99→  const seconds = Math.floor((Date.now() - new Date(iso).getTime()) / 1000)
   100→  if (seconds < 60) return 'just now'
   101→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   102→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   103→  return `${Math.floor(seconds / 86400)}d ago`
   104→}
