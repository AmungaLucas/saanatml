    77→// ── Helpers ─────────────────────────────────────────────────

    79→const slugify = (text: string) => text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, '')

    81→const fmtDate = (iso: string) =>
    82→  new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    83→
    84→const fmtDateTime = (iso: string) =>
    85→  new Date(iso).toLocaleDateString('en-KE', {
    86→    day: 'numeric', month: 'short', year: 'numeric',
    87→    hour: '2-digit', minute: '2-digit',
    88→  })
    89→
    90→const fmtEventDate = (iso: string, endDate: string | null) => {
    91→  const d = new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short' })
    92→  if (endDate) {
    93→    const e = new Date(endDate).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    94→    return `${d} – ${e}`
    95→  }
    96→  return d
    97→}
