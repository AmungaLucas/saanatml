    77→    77→const toDatetimeLocal = (iso: string) => {
    78→    78→  if (!iso) return ''
    79→    79→  const d = new Date(iso)
    80→    80→  const pad = (n: number) => String(n).padStart(2, '0')
    81→    81→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    82→    82→}
