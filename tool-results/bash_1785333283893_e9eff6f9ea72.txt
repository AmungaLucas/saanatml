   372	export function NewsletterCTA({ variant = 'inline' }: { variant?: 'inline' | 'hero' }) {
   373	  const [name, setName] = useState('')
   374	  const [email, setEmail] = useState('')
   375	  const [status, setStatus] = useState<'idle' | 'loading' | 'success' | 'error'>('idle')
   376	
   377	  async function handleSubmit(e: React.FormEvent) {
   378	    e.preventDefault()
   379	    if (!email.trim()) return
   380	    setStatus('loading')
   381	    try {
   382	      const res = await fetch('/api/newsletter', {
   383	        method: 'POST',
   384	        headers: { 'Content-Type': 'application/json' },
   385	        body: JSON.stringify({ name: name || '', email }),
   386	      })
   387	      if (res.ok) {
   388	        setStatus('success')
   389	        setName('')
   390	        setEmail('')
   391	      } else {
   392	        setStatus('error')
   393	      }
   394	    } catch {
   395	      setStatus('error')
   396	    }
   397	  }
   398	
   399	  if (status === 'success') {
   400	    return (
   401	      <div className={`rounded-2xl p-8 md:p-12 text-center ${variant === 'hero' ? 'bg-primary text-primary-foreground' : 'bg-primary/5 border border-primary/20'}`}>
   402	        <div className="flex items-center justify-center gap-2 mb-3">
   403	          <Check className="h-5 w-5" />
   404	          <span className="font-serif text-xl font-bold">You&rsquo;re subscribed!</span>
   405	        </div>
   406	        <p className="text-sm opacity-80 max-w-md mx-auto">
   407	          Welcome to the Sanaa community. Look out for our next edition in your inbox.
   408	        </p>
   409	      </div>
   410	    )
   411	  }
   412	
   413	  if (variant === 'hero') {
   414	    return (
   415	      <div className="bg-primary text-primary-foreground rounded-2xl p-8 md:p-12 text-center">
   416	        <h2 className="font-serif text-2xl md:text-3xl font-bold mb-3">This Week in East African Arts</h2>
   417	        <p className="text-primary-foreground/80 max-w-lg mx-auto mb-6 text-sm md:text-base">
   418	          Get our weekly newsletter with curated event picks, new reviews, and exclusive content from the East African art scene.
   419	        </p>
   420	        <form onSubmit={handleSubmit} className="flex flex-col sm:flex-row gap-3 max-w-md mx-auto">
   421	          <input
   422	            placeholder="Your name (optional)"
   423	            value={name}
   424	            onChange={e => setName(e.target.value)}
   425	            className="flex-1 px-4 py-2.5 rounded-lg bg-primary-foreground/10 border border-primary-foreground/20 text-sm text-primary-foreground placeholder:text-primary-foreground/50 focus:outline-none focus:ring-1 focus:ring-primary-foreground/50"
   426	          />
   427	          <input
   428	            placeholder="Your email address"
   429	            required
   430	            value={email}
   431	            onChange={e => setEmail(e.target.value)}
   432	            className="flex-1 px-4 py-2.5 rounded-lg bg-primary-foreground/10 border border-primary-foreground/20 text-sm text-primary-foreground placeholder:text-primary-foreground/50 focus:outline-none focus:ring-1 focus:ring-primary-foreground/50"
   433	          />
   434	          <button
   435	            type="submit"
   436	            disabled={status === 'loading'}
   437	            className="px-6 py-2.5 rounded-lg bg-primary-foreground text-primary font-mono text-xs font-semibold hover:bg-primary-foreground/90 transition-colors shrink-0 disabled:opacity-50"
   438	          >
   439	            {status === 'loading' ? 'Subscribing...' : 'Subscribe'}
   440	          </button>
   441	        </form>
   442	      </div>
   443	    )
   444	  }
   445	
   446	  return (
   447	    <div className="mt-12 p-6 rounded-xl bg-primary/5 border border-primary/20">
   448	      <h3 className="font-serif font-bold text-lg mb-2 flex items-center gap-2">
   449	        <Mail className="h-4 w-4 text-primary" />
   450	        Enjoyed this story?
   451	      </h3>
   452	      <p className="text-sm text-muted-foreground mb-4">Get more like this delivered to your inbox every week.</p>
   453	      <form onSubmit={handleSubmit} className="flex gap-2">
   454	        <Input
   455	          placeholder="Your name (optional)"
   456	          value={name}
   457	          onChange={e => setName(e.target.value)}
   458	          className="max-w-[200px] h-9 text-sm"
   459	        />
   460	        <Input
   461	          placeholder="Your email address"
   462	          required
   463	          value={email}
   464	          onChange={e => setEmail(e.target.value)}
   465	          className="flex-1 h-9 text-sm"
   466	        />
   467	        <Button
   468	          type="submit"
   469	          size="sm"
   470	          disabled={status === 'loading'}
   471	          className="bg-primary text-primary-foreground h-9 px-4 text-xs font-mono"
   472	        >
