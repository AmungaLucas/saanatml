    89→    89→const toDatetimeLocal = (iso: string) => {
    90→    90→  if (!iso) return ''
    91→    91→  const d = new Date(iso)
    92→    92→  const pad = (n: number) => String(n).padStart(2, '0')
    93→    93→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    94→    94→}
