     1	'use client'
     2	
     3	import { useEffect, useState } from 'react'
     4	
     5	export function ReadingProgress() {
     6	  const [progress, setProgress] = useState(0)
     7	
     8	  useEffect(() => {
     9	    const handleScroll = () => {
    10	      const scrollTop = window.scrollY
    11	      const docHeight = document.documentElement.scrollHeight - window.innerHeight
    12	      if (docHeight > 0) {
    13	        setProgress((scrollTop / docHeight) * 100)
    14	      }
    15	    }
    16	    window.addEventListener('scroll', handleScroll, { passive: true })
    17	    handleScroll()
    18	    return () => window.removeEventListener('scroll', handleScroll)
    19	  }, [])
    20	
    21	  return (
    22	    <div
    23	      className="reading-progress"
    24	      style={{ width: `${Math.min(progress, 100)}%` }}
    25	      role="progressbar"
    26	      aria-valuenow={Math.round(progress)}
    27	      aria-valuemin={0}
    28	      aria-valuemax={100}
    29	      aria-label="Reading progress"
    30	    />
    31	  )
    32	}

     9	export function StoryGrid() {

     9	export function HeroCarousel() {
    10	  const { featuredArticles, openArticle } = useStore()

    12	
    13	  const featured = featuredArticles.length > 0 ? featuredArticles : useStore.getState().articles.slice(0, 3)
    14

    24	
    25	  const article = featured[current]
    26	
    27	  return (
    28	    <section className="relative">
