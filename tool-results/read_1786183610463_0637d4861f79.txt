    66→// ── Helpers ─────────────────────────────────────────────────

    68→const slugify = (text: string) => text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-+$/, '')

    70→const toDatetimeLocal = (iso: string) => {
    71→  if (!iso) return ''
    72→  const d = new Date(iso)
    73→  const pad = (n: number) => String(n).padStart(2, '0')
    74→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    75→}

    77→const fmtDate = (iso: string) => new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric' })
    78→const fmtDateTime = (iso: string) => new Date(iso).toLocaleDateString('en-KE', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
